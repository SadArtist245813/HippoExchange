//Function used throughout program to find a user based on SessionID
function getSessionUser(SessionID, callback) {
    let sql = 'SELECT UserID FROM tblSessions WHERE SessionID = ? AND CreatedDateTime > DATE_SUB(NOW(), INTERVAL 24 HOUR)'
    let values = [SessionID]
    hippoExchange.query(sql, values, (err, rows) => {
        if(err || rows.length === 0){
            return callback(null)
        }

        callback(rows[0].UserID)
    })
}

/**
 * Route #1: Creates a new user account.
 * Validates the input, hashes the password, and inserts the user into the database.
 */
app.post('/user', (req, res, next) => {
    let {email, password, firstName, lastName} = req.body

    if(!email || !password || !firstName || !lastName){
        return res.status(400).json({status:"Failed"})
    }

    if(!emailRegex.test(email) || !passwordRegex.test(password)){
        return res.status(400).json({status:"Failed"})
    }

    try{
        bcrypt.hash(password, 10, (err, hash) => {
            if(err){
                return res.status(500).json({status:"Failed"})
            }
            let sql = 'INSERT INTO tblUsers (Email, FirstName, LastName, Password, CreatedDateTime, LastUsedDateTime) VALUES (?, ?, ?, ?, NOW(), NOW())'
            let values = [email, firstName, lastName, hash]
            hippoExchange.query(sql, values, (err, result) => {
                if(err){
                    return res.status(500).json({status:"Failed"})
                }

                res.json({status:"Success"})
            })
        })
    }catch(error){
        console.error('Unexpected error:', error)
        res.status(500).json({ status: 'Failed', error: 'Unexpected error occurred' })
    }
})
